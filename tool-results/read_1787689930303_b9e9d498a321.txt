   254→// ─── Create Wallet Flow ──────────────────────────────────────────────
   255→function CreateWalletScreen() {
   256→  const navigate = useWalletStore((s) => s.navigate);
   257→  const setTempSeedPhrase = useWalletStore((s) => s.setTempSeedPhrase);
   258→  const setWalletCreated = useWalletStore((s) => s.setWalletCreated);
   259→  const setLoading = useWalletStore((s) => s.setLoading);
   260→  const isLoading = useWalletStore((s) => s.isLoading);
   261→  const addToast = useWalletStore((s) => s.addToast);
   262→  const setShowOnboarding = useWalletStore((s) => s.setShowOnboarding);
   263→
   264→  const [step, setStep] = useState<0 | 1 | 2 | 3 | 4>(0);
   265→  const [seedPhrase, setSeedPhrase] = useState<string[]>([]);
   266→  const [verifyWords, setVerifyWords] = useState<Record<number, string>>({});
   267→  const [verifyErrors, setVerifyErrors] = useState<Set<number>>(new Set());
   268→  const [password, setPassword] = useState('');
   269→  const [confirmPassword, setConfirmPassword] = useState('');
   270→  const [showPassword, setShowPassword] = useState(false);
   271→  const [walletAddress, setWalletAddress] = useState('');
   272→
   273→  const handleGenerate = useCallback(async () => {
   274→    setLoading(true);
   275→    try {
   276→      const phrase = await generateSeedPhrase();
   277→      const words = phrase.split(' ');
   278→      setSeedPhrase(words);
   279→      setTempSeedPhrase(phrase);
   280→      setStep(1);
   281→    } catch {
   282→      addToast('Failed to generate seed phrase', 'error');
   283→    }
   284→    setLoading(false);
   285→  }, [setLoading, setTempSeedPhrase, addToast]);
   286→
   287→  const handleVerifyNext = useCallback(() => {
   288→    const errors = new Set<number>();
   289→    const challengeIndices = [1, 4, 8, 11];
   290→    challengeIndices.forEach((idx) => {
   291→      if (verifyWords[idx]?.toLowerCase().trim() !== seedPhrase[idx]?.toLowerCase()) {
   292→        errors.add(idx);
   293→      }
   294→    });
   295→    setVerifyErrors(errors);
   296→    if (errors.size === 0) {
   297→      setStep(3);
   298→    } else {
   299→      addToast('Some words are incorrect. Please try again.', 'error');
   300→    }
   301→  }, [verifyWords, seedPhrase, addToast]);
   302→
   303→  const handleCreateWallet = useCallback(async () => {
   304→    if (password.length < 8) {
   305→      addToast('Password must be at least 8 characters', 'error');
   306→      return;
   307→    }
   308→    if (password !== confirmPassword) {
   309→      addToast('Passwords do not match', 'error');
   310→      return;
   311→    }
   312→    setLoading(true);
   313→    try {
   314→      const result = await createEncryptedWallet(password);
   315→      setWalletAddress(result.address);
   316→      setWalletCreated(result.address, result.encryptedPrivateKey);
   317→      setStep(4);
   318→      addToast('Wallet created successfully!', 'success');
   319→    } catch (err) {
   320→      console.error('[QFS] createEncryptedWallet failed:', err);
   321→      addToast('Failed to create wallet', 'error');
   322→    }
   323→    setLoading(false);
   324→  }, [password, confirmPassword, setWalletCreated, setLoading, addToast]);
   325→
   326→  return (
   327→    <motion.div
   328→      key="create-wallet"
   329→      variants={pageVariants}
   330→      initial="initial"
   331→      animate="animate"
   332→      exit="exit"
   333→      transition={pageTransition}
   334→      className="min-h-screen flex flex-col"
   335→    >
   336→      <ScreenHeader title="Create Wallet" onBack={step === 0 ? () => setShowOnboarding(true) || navigate('create-wallet') : () => setStep((s) => Math.max(0, (s - 1)) as 0)} />
   337→
   338→      {/* Progress bar */}
   339→      <div className="px-4 mb-2">
   340→        <div className="flex gap-1.5">
   341→          {[0, 1, 2, 3, 4].map((i) => (
   342→            <div key={i} className={`h-1 flex-1 rounded-full transition-colors duration-300 ${i <= step ? 'bg-primary' : 'bg-border'}`} />
   343→          ))}
   344→        </div>
   345→        <p className="text-xs text-muted-foreground mt-2 text-center">Step {step + 1} of 5</p>
   346→      </div>
   347→
   348→      <div className="flex-1 px-4 pb-8 overflow-y-auto">
   349→        <AnimatePresence mode="wait">
   350→          {/* Step 0: Info */}
   351→          {step === 0 && (
   352→            <motion.div key="step0" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }} className="flex flex-col items-center pt-12">
   353→              <div className="w-20 h-20 rounded-2xl bg-primary/10 border border-primary/15 flex items-center justify-center mb-6">
   354→                <Wallet className="size-10 text-primary" />
   355→              </div>
   356→              <h2 className="text-xl font-semibold mb-3">Create a New Wallet</h2>
   357→              <p className="text-muted-foreground text-sm text-center max-w-xs mb-8">
   358→                Your wallet will be secured by a 12-word recovery phrase. Keep it safe, it&apos;s the only way to recover your funds.
   359→              </p>
   360→              <div className="w-full max-w-xs flex flex-col gap-3">
   361→                {['Non-custodial, only you control your keys', 'Multi-chain support from day one', 'Built-in staking & swap features'].map((feature, i) => (
   362→                  <div key={i} className="flex items-center gap-3 text-sm text-muted-foreground">
   363→                    <div className="w-5 h-5 rounded-full bg-primary/10 flex items-center justify-center shrink-0">
   364→                      <Check className="size-3 text-primary" />
   365→                    </div>
   366→                    {feature}
   367→                  </div>
   368→                ))}
   369→              </div>
   370→              <button
   371→                onClick={handleGenerate}
   372→                disabled={isLoading}
   373→                className="mt-10 w-full max-w-xs h-12 rounded-xl emerald-gradient text-black font-semibold hover:opacity-90 transition-opacity disabled:opacity-50"
   374→              >
   375→                {isLoading ? 'Generating...' : 'Generate Recovery Phrase'}
   376→              </button>
   377→            </motion.div>
   378→          )}
   379→
   380→          {/* Step 1: Show seed phrase */}
   381→          {step === 1 && (
   382→            <motion.div key="step1" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }} className="pt-4">
   383→              <div className="flex items-center gap-2 mb-2 p-3 rounded-xl bg-qfs-red/10 border border-qfs-red/20">
   384→                <AlertTriangle className="size-4 text-qfs-red shrink-0" />
   385→                <p className="text-xs text-qfs-red">Never share your recovery phrase with anyone. Store it offline in a secure place.</p>
   386→              </div>
   387→              <h2 className="text-lg font-semibold mb-1">Your Recovery Phrase</h2>
   388→              <p className="text-sm text-muted-foreground mb-4">Write down these 12 words in order.</p>
   389→              <div className="grid grid-cols-3 gap-2 mb-6">
   390→                {seedPhrase.map((word, i) => (
   391→                  <div key={i} className="bg-secondary rounded-lg px-3 py-2.5 flex items-center gap-2">
   392→                    <span className="text-xs text-muted-foreground w-4">{i + 1}.</span>
   393→                    <span className="text-sm font-medium">{word}</span>
   394→                  </div>
   395→                ))}
   396→              </div>
   397→              <button
   398→                onClick={() => setStep(2)}
   399→                className="w-full h-12 rounded-xl emerald-gradient text-black font-semibold hover:opacity-90 transition-opacity"
   400→              >
   401→                I&apos;ve Saved My Phrase — Continue
   402→              </button>
   403→            </motion.div>
   404→          )}
   405→
   406→          {/* Step 2: Verify seed phrase */}
   407→          {step === 2 && (
   408→            <motion.div key="step2" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }} className="pt-4">
   409→              <h2 className="text-lg font-semibold mb-1">Verify Your Phrase</h2>
   410→              <p className="text-sm text-muted-foreground mb-6">Select the correct word for each position.</p>
   411→              <div className="flex flex-col gap-4 mb-6">
   412→                {[1, 4, 8, 11].map((idx) => (
   413→                  <div key={idx}>
   414→                    <label className="text-xs text-muted-foreground mb-1.5 block">Word #{idx + 1}</label>
   415→                    <Input
   416→                      placeholder={`Enter word ${idx + 1}`}
   417→                      value={verifyWords[idx] || ''}
   418→                      onChange={(e) => {
   419→                        setVerifyWords((p) => ({ ...p, [idx]: e.target.value }));
   420→                        setVerifyErrors((p) => { const n = new Set(p); n.delete(idx); return n; });
   421→                      }}
   422→                      className={verifyErrors.has(idx) ? 'border-qfs-red' : ''}
   423→                    />
   424→                  </div>
   425→                ))}
   426→              </div>
   427→              <button
   428→                onClick={handleVerifyNext}
   429→                className="w-full h-12 rounded-xl emerald-gradient text-black font-semibold hover:opacity-90 transition-opacity"
   430→              >
   431→                Verify

   433→            </motion.div>
   434→          )}
   435→
   436→          {/* Step 3: Set password */}
   437→          {step === 3 && (
   438→            <motion.div key="step3" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }} className="pt-4">
   439→              <div className="flex items-center gap-3 mb-6">
