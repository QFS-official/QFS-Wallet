   273→// ─── Create Wallet Flow ──────────────────────────────────────────────
   274→function CreateWalletScreen() {
   275→  const navigate = useWalletStore((s) => s.navigate);
   276→  const setTempSeedPhrase = useWalletStore((s) => s.setTempSeedPhrase);
   277→  const setWalletCreated = useWalletStore((s) => s.setWalletCreated);
   278→  const setLoading = useWalletStore((s) => s.setLoading);
   279→  const isLoading = useWalletStore((s) => s.isLoading);
   280→  const addToast = useWalletStore((s) => s.addToast);
   281→  const setShowOnboarding = useWalletStore((s) => s.setShowOnboarding);
   282→
   283→  const [step, setStep] = useState<0 | 1 | 2 | 3 | 4>(0);
   284→  const [seedPhrase, setSeedPhrase] = useState<string[]>([]);
   285→  const [verifyWords, setVerifyWords] = useState<Record<number, string>>({});
   286→  const [verifyErrors, setVerifyErrors] = useState<Set<number>>(new Set());
   287→  const [password, setPassword] = useState('');
   288→  const [confirmPassword, setConfirmPassword] = useState('');
   289→  const [showPassword, setShowPassword] = useState(false);
   290→  const [walletAddress, setWalletAddress] = useState('');
   291→
   292→  const handleGenerate = useCallback(async () => {
   293→    setLoading(true);
   294→    try {
   295→      const phrase = await generateSeedPhrase();
   296→      const words = phrase.split(' ');
   297→      setSeedPhrase(words);
   298→      setTempSeedPhrase(phrase);
   299→      setStep(1);
   300→    } catch {
   301→      addToast('Failed to generate seed phrase', 'error');
   302→    }
   303→    setLoading(false);
   304→  }, [setLoading, setTempSeedPhrase, addToast]);
   305→
   306→  const handleVerifyNext = useCallback(() => {
   307→    const errors = new Set<number>();
   308→    [1, 4, 8, 11].forEach((idx) => {
   309→      if (verifyWords[idx]?.toLowerCase().trim() !== seedPhrase[idx]?.toLowerCase()) errors.add(idx);
   310→    });
   311→    setVerifyErrors(errors);
   312→    if (errors.size === 0) { setStep(3); } else { addToast('Some words are incorrect.', 'error'); }
   313→  }, [verifyWords, seedPhrase, addToast]);
   314→
   315→  const handleCreateWallet = useCallback(async () => {
   316→    if (password.length < 8) { addToast('Password must be at least 8 characters', 'error'); return; }
   317→    if (password !== confirmPassword) { addToast('Passwords do not match', 'error'); return; }
   318→    setLoading(true);
   319→    try {
   320→      const result = await createEncryptedWallet(password);
   321→      setWalletAddress(result.address);
   322→      setWalletCreated(result.address, result.encryptedPrivateKey);
   323→      setStep(4);
   324→      addToast('Wallet created successfully!', 'success');
   325→    } catch (err) {
   326→      console.error('[QFS] createEncryptedWallet failed:', err);
   327→      addToast('Failed to create wallet', 'error');
   328→    }
   329→    setLoading(false);
   330→  }, [password, confirmPassword, setWalletCreated, setLoading, addToast]);
   331→
   332→  return (
   333→    <motion.div key="create-wallet" variants={pageVariants} initial="initial" animate="animate" exit="exit" transition={pageTransition} className="min-h-screen flex flex-col bg-[#181A20]">
   334→      <ScreenHeader title="Create Wallet" onBack={step === 0 ? () => setShowOnboarding(true) || navigate('create-wallet') : () => setStep((s) => Math.max(0, (s - 1)) as 0)} />
   335→
   336→      {/* Progress bar */}
   337→      <div className="px-4 mb-2">
   338→        <div className="flex gap-1.5">
   339→          {[0, 1, 2, 3, 4].map((i) => (
   340→            <div key={i} className={`h-1 flex-1 rounded-full transition-colors duration-300 ${i <= step ? 'bg-[#F0B90B]' : 'bg-[#2B3139]'}`} />
   341→          ))}
   342→        </div>
   343→        <p className="text-xs text-[#848E9C] mt-2 text-center">Step {step + 1} of 5</p>
   344→      </div>
   345→
   346→      <div className="flex-1 px-4 pb-8 overflow-y-auto">
   347→        <AnimatePresence mode="wait">
   348→          {step === 0 && (
   349→            <motion.div key="step0" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }} className="flex flex-col items-center pt-12">
   350→              <div className="w-20 h-20 rounded-2xl bg-[#2B3139] flex items-center justify-center mb-6">
   351→                <Wallet className="size-10 text-[#F0B90B]" />
   352→              </div>
   353→              <h2 className="text-xl font-semibold mb-3 text-[#EAECEF]">Create a New Wallet</h2>
   354→              <p className="text-[#848E9C] text-sm text-center max-w-xs mb-8">
   355→                Your wallet will be secured by a 12-word recovery phrase. Keep it safe.
   356→              </p>
   357→              <div className="w-full max-w-xs flex flex-col gap-3">
   358→                {['Non-custodial — only you control your keys', 'Multi-chain support from day one', 'Built-in staking & swap features'].map((feature, i) => (
   359→                  <div key={i} className="flex items-center gap-3 text-sm text-[#848E9C]">
   360→                    <div className="w-5 h-5 rounded-full bg-[#F0B90B]/10 flex items-center justify-center shrink-0">
   361→                      <Check className="size-3 text-[#F0B90B]" />
   362→                    </div>
   363→                    {feature}
   364→                  </div>
   365→                ))}
   366→              </div>
   367→              <button onClick={handleGenerate} disabled={isLoading} className="mt-10 w-full max-w-xs h-[52px] rounded-xl bnb-btn-primary text-base">
   368→                {isLoading ? 'Generating...' : 'Generate Recovery Phrase'}
   369→              </button>
   370→            </motion.div>
   371→          )}
   372→
   373→          {step === 1 && (
   374→            <motion.div key="step1" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }} className="pt-4">
   375→              <div className="flex items-center gap-2 mb-4 p-3 rounded-xl bg-[#F6465D]/10 border border-[#F6465D]/20">
   376→                <AlertTriangle className="size-4 text-[#F6465D] shrink-0" />

   392→            </motion.div>
   393→          )}
   394→
   395→          {step === 2 && (
   396→            <motion.div key="step2" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }} className="pt-4">
   397→              <h2 className="text-lg font-semibold mb-1 text-[#EAECEF]">Verify Your Phrase</h2>
   398→              <p className="text-sm text-[#848E9C] mb-6">Select the correct word for each position.</p>

'border-[#F6465D]' : ''} />

  1933→function QFSWallet() {
  1934→  const currentScreen = useWalletStore((s) => s.currentScreen);
  1935→  const isWalletCreated = useWalletStore((s) => s.isWalletCreated);
  1936→  const isWalletLocked = useWalletStore((s) => s.isWalletLocked);
  1937→  const showOnboarding = useWalletStore((s) => s.showOnboarding);
  1938→  const initialize = useWalletStore((s) => s.initialize);
  1939→
  1940→  useEffect(() => { initialize(); }, [initialize]);
  1941→
  1942→  const showNav = isWalletCreated && !isWalletLocked && !['create-wallet', 'import-wallet', 'seed-verify'].includes(currentScreen);
  1943→
  1944→  const renderScreen = () => {
  1945→    if (isWalletCreated && isWalletLocked) return <WalletLockScreen />;
  1946→    if (showOnboarding && !isWalletCreated) return <OnboardingScreen />;
  1947→    switch (currentScreen) {
  1948→      case 'create-wallet': return <CreateWalletScreen />;
  1949→      case 'import-wallet': return <ImportWalletScreen />;
  1950→      case 'dashboard': return <DashboardScreen />;
  1951→      case 'send': return <SendScreen />;
  1952→      case 'receive': return <ReceiveScreen />;
  1953→      case 'swap': return <SwapScreen />;
  1954→      case 'staking': return <StakingScreen />;
  1955→      case 'dapps': return <DAppsScreen />;
  1956→      case 'settings': return <SettingsScreen />;
  1957→      case 'wallet': return <WalletScreen />;
  1958→      case 'networks': return <NetworksScreen />;
  1959→      default: return <DashboardScreen />;
  1960→    }
  1961→  };
  1962→
  1963→  return (
  1964→    <div className="min-h-screen bg-[#181A20] max-w-md lg:max-w-lg mx-auto relative">
  1965→      <ToastContainer />
  1966→      <AnimatePresence mode="wait">
  1967→        {renderScreen()}
  1968→      </AnimatePresence>
  1969→      {showNav && <BottomNav />}
  1970→    </div>
  1971→  );
  1972→}
