   280→// ─── Create Wallet Flow ──────────────────────────────────────────────
   281→function CreateWalletScreen() {
   282→  const navigate = useWalletStore((s) => s.navigate);
   283→  const setTempSeedPhrase = useWalletStore((s) => s.setTempSeedPhrase);
   284→  const setWalletCreated = useWalletStore((s) => s.setWalletCreated);
   285→  const setLoading = useWalletStore((s) => s.setLoading);
   286→  const isLoading = useWalletStore((s) => s.isLoading);
   287→  const addToast = useWalletStore((s) => s.addToast);
   288→  const setShowOnboarding = useWalletStore((s) => s.setShowOnboarding);
   289→
   290→  const [step, setStep] = useState<0 | 1 | 2 | 3 | 4>(0);
   291→  const [seedPhrase, setSeedPhrase] = useState<string[]>([]);
   292→  const [verifyWords, setVerifyWords] = useState<Record<number, string>>({});
   293→  const [verifyErrors, setVerifyErrors] = useState<Set<number>>(new Set());
   294→  const [password, setPassword] = useState('');
   295→  const [confirmPassword, setConfirmPassword] = useState('');
   296→  const [showPassword, setShowPassword] = useState(false);
   297→  const [walletAddress, setWalletAddress] = useState('');
   298→
   299→  const handleGenerate = useCallback(async () => {
   300→    setLoading(true);
   301→    try {
   302→      const phrase = await generateSeedPhrase();
   303→      const words = phrase.split(' ');
   304→      setSeedPhrase(words);
   305→      setTempSeedPhrase(phrase);
   306→      setStep(1);
   307→    } catch {
   308→      addToast('Failed to generate seed phrase', 'error');
   309→    }
   310→    setLoading(false);
   311→  }, [setLoading, setTempSeedPhrase, addToast]);
   312→
   313→  const handleVerifyNext = useCallback(() => {
   314→    const errors = new Set<number>();
   315→    const challengeIndices = [1, 4, 8, 11];
   316→    challengeIndices.forEach((idx) => {
   317→      if (verifyWords[idx]?.toLowerCase().trim() !== seedPhrase[idx]?.toLowerCase()) {
   318→        errors.add(idx);
   319→      }
   320→    });
   321→    setVerifyErrors(errors);
   322→    if (errors.size === 0) {
   323→      setStep(3);
   324→    } else {
   325→      addToast('Some words are incorrect. Please try again.', 'error');
   326→    }
   327→  }, [verifyWords, seedPhrase, addToast]);
   328→
   329→  const handleCreateWallet = useCallback(async () => {
   330→    if (password.length < 8) {
   331→      addToast('Password must be at least 8 characters', 'error');
   332→      return;
   333→    }
   334→    if (password !== confirmPassword) {
   335→      addToast('Passwords do not match', 'error');
   336→      return;
   337→    }
   338→    setLoading(true);
   339→    try {
   340→      const result = await createEncryptedWallet(password);
   341→      setWalletAddress(result.address);
   342→      setWalletCreated(result.address, result.encryptedPrivateKey);
   343→      setStep(4);
   344→      addToast('Wallet created successfully!', 'success');
   345→    } catch (err) {
   346→      console.error('[QFS] createEncryptedWallet failed:', err);
   347→      addToast('Failed to create wallet', 'error');
   348→    }
   349→    setLoading(false);
   350→  }, [password, confirmPassword, setWalletCreated, setLoading, addToast]);
   351→
   352→  return (
   353→    <motion.div
   354→      key="create-wallet"
   355→      variants={pageVariants}
   356→      initial="initial"
   357→      animate="animate"
   358→      exit="exit"
   359→      transition={pageTransition}
   360→      className="min-h-screen flex flex-col"
   361→    >
   362→      <ScreenHeader title="Create Wallet" onBack={step === 0 ? () => setShowOnboarding(true) || navigate('create-wallet') : () => setStep((s) => Math.max(0, (s - 1)) as 0)} />
   363→
   364→      {/* Progress bar */}
   365→      <div className="px-4 mb-2">
   366→        <div className="flex gap-1.5">
   367→          {[0, 1, 2, 3, 4].map((i) => (
   368→            <div key={i} className={`h-1 flex-1 rounded-full transition-colors duration-300 ${i <= step ? 'bg-primary' : 'bg-border'}`} />
   369→          ))}
   370→        </div>
   371→        <p className="text-xs text-muted-foreground mt-2 text-center">Step {step + 1} of 5</p>
   372→      </div>
   373→
   374→      <div className="flex-1 px-4 pb-8 overflow-y-auto">
   375→        <AnimatePresence mode="wait">
   376→          {/* Step 0: Info */}
   377→          {step === 0 && (
   378→            <motion.div key="step0" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }} className="flex flex-col items-center pt-12">
   379→              <div className="w-20 h-20 rounded-2xl bg-primary/10 border border-primary/15 flex items-center justify-center mb-6">
   380→                <Wallet className="size-10 text-accent" />
   381→              </div>
   382→              <h2 className="text-xl font-semibold mb-3">Create a New Wallet</h2>
   383→              <p className="text-muted-foreground text-sm text-center max-w-xs mb-8">
   384→                Your wallet will be secured by a 12-word recovery phrase. Keep it safe, it&apos;s the only way to recover your funds.
   385→              </p>
   386→              <div className="w-full max-w-xs flex flex-col gap-3">
   387→                {['Non-custodial, only you control your keys', 'Multi-chain support from day one', 'Built-in staking & swap features'].map((feature, i) => (
   388→                  <div key={i} className="flex items-center gap-3 text-sm text-muted-foreground">
   389→                    <div className="w-5 h-5 rounded-full bg-primary/10 flex items-center justify-center shrink-0">
   390→                      <Check className="size-3 text-accent" />
   391→                    </div>
   392→                    {feature}
   393→                  </div>
   394→                ))}
   395→              </div>
   396→              <button
   397→                onClick={handleGenerate}
   398→                disabled={isLoading}
   399→                className="mt-10 w-full max-w-xs h-12 rounded-xl blue-gold-gradient text-white font-semibold hover:opacity-90 transition-opacity disabled:opacity-50"
   400→              >
   401→                {isLoading ? 'Generating...' : 'Generate Recovery Phrase'}
   402→              </button>
   403→            </motion.div>
   404→          )}
   405→
   406→          {/* Step 1: Show seed phrase */}
   407→          {step === 1 && (
   408→            <motion.div key="step1" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }} className="pt-4">
   409→              <div className="flex items-center gap-2 mb-2 p-3 rounded-xl bg-qfs-red/10 border border-qfs-red/20">
   410→                <AlertTriangle className="size-4 text-qfs-red shrink-0" />
   411→                <p className="text-xs text-qfs-red">Never share your recovery phrase with anyone. Store it offline in a secure place.</p>
   412→              </div>
   413→              <h2 className="text-lg font-semibold mb-1">Your Recovery Phrase</h2>
   414→              <p className="text-sm text-muted-foreground mb-4">Write down these 12 words in order.</p>
   415→              <div className="grid grid-cols-3 gap-2 mb-6">
   416→                {seedPhrase.map((word, i) => (
   417→                  <div key={i} className="bg-secondary rounded-lg px-3 py-2.5 flex items-center gap-2">
   418→                    <span className="text-xs text-muted-foreground w-4">{i + 1}.</span>
   419→                    <span className="text-sm font-medium">{word}</span>
   420→                  </div>
   421→                ))}
   422→              </div>
   423→              <button
   424→                onClick={() => setStep(2)}
   425→                className="w-full h-12 rounded-xl blue-gold-gradient text-white font-semibold hover:opacity-90 transition-opacity"
   426→              >
   427→                I&apos;ve Saved My Phrase — Continue
   428→              </button>
   429→            </motion.div>
   430→          )}
   431→
   432→          {/* Step 2: Verify seed phrase */}
   433→          {step === 2 && (
   434→            <motion.div key="step2" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }} className="pt-4">
   435→              <h2 className="text-lg font-semibold mb-1">Verify Your Phrase</h2>
   436→              <p className="text-sm text-muted-foreground mb-6">Select the correct word for each position.</p>
   437→              <div className="flex flex-col gap-4 mb-6">
   438→                {[1, 4, 8, 11].map((idx) => (
   439→                  <div key={idx}>
   440→                    <label className="text-xs text-muted-foreground mb-1.5 block">Word #{idx + 1}</label>
   441→                    <Input
   442→                      placeholder={`Enter word ${idx + 1}`}
   443→                      value={verifyWords[idx] || ''}
   444→                      onChange={(e) => {
   445→                        setVerifyWords((p) => ({ ...p, [idx]: e.target.value }));
   446→                        setVerifyErrors((p) => { const n = new Set(p); n.delete(idx); return n; });
   447→                      }}
   448→                      className={verifyErrors.has(idx) ? 'border-qfs-red' : ''}
   449→                    />
   450→                  </div>
   451→                ))}
   452→              </div>
   453→              <button
   454→                onClick={handleVerifyNext}
   455→                className="w-full h-12 rounded-xl blue-gold-gradient text-white font-semibold hover:opacity-90 transition-opacity"
   456→              >
   457→                Verify
   458→              </button>
   459→            </motion.div>
   460→          )}

   545→function ImportWalletScreen() {
   546→  const navigate = useWalletStore((s) => s.navigate);
   547→  const setWalletCreated = useWalletStore((s) => s.setWalletCreated);
   548→  const setLoading = useWalletStore((s) => s.setLoading);
   549→  const isLoading = useWalletStore((s) => s.isLoading);
   550→  const addToast = useWalletStore((s) => s.addToast);
   551→
   552→  const [mode, setMode] = useState<'seed' | 'key'>('seed');
   553→  const [seedInput, setSeedInput] = useState('');
   554→  const [keyInput, setKeyInput] = useState('');
   555→  const [password, setPassword] = useState('');
   556→  const [showPassword, setShowPassword] = useState(false);
   557→  const [step, setStep] = useState<0 | 1>(0);
   558→
   559→  const handleImport = useCallback(async () => {
   560→    if (mode === 'seed') {
   561→      const words = seedInput.trim().split(/\s+/);
   562→      if (words.length !== 12) {
   563→        addToast('Seed phrase must be exactly 12 words', 'error');
   564→        return;
   565→      }
   566→      if (!validateSeedPhrase(seedInput.trim())) {
   567→        addToast('Invalid seed phrase. Please check your words.', 'error');
   568→        return;
   569→      }
   570→    }
   571→    if (password.length < 8) {
   572→      addToast('Password must be at least 8 characters', 'error');
   573→      return;
   574→    }
   575→    setStep(1);
   576→  }, [mode, seedInput, password, addToast]);
   577→
   578→  const handleConfirm = useCallback(async () => {
   579→    setLoading(true);
   580→    try {
   581→      const mnemonic = mode === 'seed' ? seedInput.trim() : undefined;
   582→      const result = await createEncryptedWallet(password, mnemonic);
   583→      setWalletCreated(result.address, result.encryptedPrivateKey);
   584→      addToast('Wallet imported successfully!', 'success');
   585→      navigate('dashboard');
   586→    } catch {
   587→      addToast('Failed to import wallet', 'error');
   588→    }
   589→    setLoading(false);
   590→  }, [mode, seedInput, password, setWalletCreated, navigate, setLoading, addToast]);
   591→
   592→  return (
   593→    <motion.div
   594→      key="import-wallet"
   595→      variants={pageVariants}
   596→      initial="initial"
   597→      animate="animate"
   598→      exit="exit"
   599→      transition={pageTransition}
   600→      className="min-h-screen flex flex-col"
   601→    >
   602→      <ScreenHeader title="Import Wallet" onBack={() => navigate('onboarding' as Screen)} />
   603→      <div className="flex-1 px-4 pb-8 overflow-y-auto">
   604→        {step === 0 ? (
   605→          <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="pt-4">
   606→            <div className="flex gap-2 mb-6">
   607→              {(['seed', 'key'] as const).map((m) => (
   608→                <button
   609→                  key={m}
   610→                  onClick={() => setMode(m)}
   611→                  className={`flex-1 h-10 rounded-lg text-sm font-medium transition-colors ${
   612→                    mode === m ? 'bg-primary text-black' : 'bg-secondary text-muted-foreground hover:text-foreground'
   613→                  }`}
   614→                >
   615→                  {m === 'seed' ? 'Seed Phrase' : 'Private Key'}
   616→                </button>
   617→              ))}
   618→            </div>
   619→
   620→            {mode === 'seed' ? (
   621→              <div>
   622→                <label className="text-xs text-muted-foreground mb-1.5 block">12-Word Recovery Phrase</label>
   623→                <textarea
   624→                  value={seedInput}
   625→                  onChange={(e) => setSeedInput(e.target.value)}
   626→                  placeholder="Enter your 12-word recovery phrase separated by spaces"
   627→                  rows={4}
   628→                  className="w-full bg-secondary border border-border rounded-lg px-3 py-2.5 text-sm placeholder:text-muted-foreground focus:outline-none focus:border-primary resize-none"

   632→              <div>
   633→                <label className="text-xs text-muted-foreground mb-1.5 block">Private Key</label>
   634→                <Input
   635→                  value={keyInput}
   636→                  onChange={(e) => setKeyInput(e.target.value)}
   637→                  placeholder="Enter your private key (hex format)"
   638→                  type="password"
   639→                />
   640→                <div className="flex items-center gap-1.5 mt-2">
   641→                  <AlertTriangle className="size-3 text-qfs-red" />
   642→                  <p className="text-xs text-muted-foreground">Importing via private key is less secure</p>
   643→                </div>
   644→              </div>
   645→            )}
   646→
   647→            <div className="mt-6">
   648→              <label className="text-xs text-muted-foreground mb-1.5 block">Encryption Password</label>
   649→              <div className="relative">
   650→                <Input
   651→                  type={showPassword ? 'text' : 'password'}

   654→                  onChange={(e) => setPassword(e.target.value)}
   655→                  className="pr-10"
